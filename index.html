  CHARTS.vendNeg = new Chart($('chartVendNeg').getContext('2d'), {

    type: 'bar',

    data: {

      labels: negLabels,

      datasets: [

        { label: 'Impactados', data: negImp, backgroundColor: negLabels.map(k => (NEG_COLORS[k]||COLORS.blue)+'CC'), borderRadius:3, borderSkipped:false, borderWidth:0 },

        { label: 'Maestro',    data: negMae, backgroundColor: 'rgba(148,163,184,.15)', borderRadius:3, borderSkipped:false, borderWidth:0 },

      ]

    },

    options: {

      indexAxis: 'y', responsive: true, maintainAspectRatio: false,

      plugins: { legend: { labels: { color: '#94A3B8', font: { size: 10 } } }, tooltip: { callbacks: { label: c => ` ${c.dataset.label}: ${c.raw}` } } },

      scales: {

        x: { grid: { color: 'rgba(148,163,184,.08)' }, ticks: { color: '#94A3B8', font:{size:10} }, border:{display:false} },

        y: { grid: { display: false }, ticks: { color: '#94A3B8', font:{size:11} }, border:{display:false} }

      }

    }

  });


  // Cobertura vs equipo

  const allVends = [...STATE.cobertura]

    .filter(r => !String(r.vendedor||'').toUpperCase().includes('TOTAL'))

    .sort((a,b) => (parseFloat(b.cobertura__||b['cobertura_']||0)) - (parseFloat(a.cobertura__||a['cobertura_']||0)));


  const cobField = allVends.length > 0 ? Object.keys(allVends[0]).find(k => k.includes('cobertura'))||'cobertura_' : 'cobertura_';


  dc('chartVendCob');

  CHARTS.vendCob = new Chart($('chartVendCob').getContext('2d'), {

    type: 'bar',

    data: {

      labels: allVends.map(x => String(x.vendedor||'').split('-')[0].trim()),

      datasets: [

        { data: allVends.map(x => parseFloat(x[cobField])||0),

          backgroundColor: allVends.map(x => String(x.vendedor||'').split('-')[0].trim() === cod ? COLORS.cyan+'CC' : 'rgba(148,163,184,.2)'),

          borderRadius:3, borderSkipped:false, borderWidth:0 },

        { type:'line', data: allVends.map(()=>75), borderColor:COLORS.red, borderWidth:1.5, borderDash:[5,3], pointRadius:0, fill:false, tension:0 }

      ]

    },

    options: {

      responsive:true, maintainAspectRatio:false,

      plugins:{legend:{display:false},tooltip:{callbacks:{label:c=>c.datasetIndex===0?pct(c.raw):'Meta: 75%'}}},

      scales:{

        x:{grid:{display:false},ticks:{color:'#94A3B8',font:{size:10}},border:{display:false}},

        y:{min:0,max:105,grid:{color:'rgba(148,163,184,.08)'},ticks:{callback:v=>v+'%',color:'#94A3B8',font:{size:10}},border:{display:false}}

      }

    }

  });


  // Alertas vendedor

  const als = [];

  if (dp > 10) als.push({ t: 'alert-red',   m: `🔴 <strong>Devoluciones elevadas:</strong> ${dp}% de la venta bruta` });

  if (v.cobertura < 65) als.push({ t: 'alert-red',   m: `🔴 <strong>Cobertura crítica:</strong> ${pct(v.cobertura)} — ${v.maestro - v.clientes_imp} clientes sin compra` });

  else if (v.cobertura < 75) als.push({ t: 'alert-amber', m: `⚠ <strong>Cobertura bajo meta:</strong> ${pct(v.cobertura)} — faltan ${v.maestro - v.clientes_imp} clientes` });

  else als.push({ t: 'alert-green', m: `✅ <strong>Cobertura en meta:</strong> ${v.clientes_imp} de ${v.maestro} clientes impactados` });

  als.push({ t: 'alert-blue', m: `ℹ <strong>Efectividad del mes:</strong> ${pct(v.efectividad)} · Rank #${STATE.vendedores.indexOf(v)+1} del equipo` });

  $('vendAlerts').innerHTML = als.map(a => `<div class="alert ${a.t}">${a.m}</div>`).join('');

}

function renderCobertura() {

  const data = STATE.cobertura.filter(r => !String(r.vendedor||'').toUpperCase().includes('TOTAL'));

  const cobField = data.length > 0 ? Object.keys(data[0]).find(k => k.includes('cobertura'))||'cobertura_' : 'cobertura_';

  const sinCompraField = data.length > 0 ? Object.keys(data[0]).find(k => k.includes('sin'))||'sin_compra' : 'sin_compra';


  $('tbCobertura').innerHTML = data

    .sort((a,b) => (parseFloat(b[cobField])||0) - (parseFloat(a[cobField])||0))

    .map(r => {

      const cob = parseFloat(r[cobField]) || 0;

      const cs = cob >= 75 ? 'up' : cob >= 60 ? 'warn' : 'down';

      return `<tr>

        <td>${r.vendedor||''}</td>

        <td style="text-align:right;color:var(--muted)">${r.clientes_maestro||0}</td>

        <td style="text-align:right">${r.impactados||0}</td>

        <td style="text-align:right"><span class="${cs}">${pct(cob)}</span></td>

        <td style="text-align:right;color:var(--red)">${r[sinCompraField]||0}</td>

      </tr>`;

    }).join('');


  const cero = STATE.clientesCero;

  $('totalCero').textContent = `${cero.total} clientes`;

  $('tbCero').innerHTML = cero.por_vendedor

    .map(r => `<tr><td>${r.vendedor}</td><td style="text-align:right;color:var(--red);font-family:'DM Mono',monospace">${r.cantidad}</td></tr>`)

    .join('');

}
